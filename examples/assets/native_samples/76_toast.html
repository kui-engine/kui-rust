§2 Lua API

-- 快捷工厂(默认 duration:info/success=3000ms,warning=4000ms,error=5000ms)
window.toast:info("Saved")
window.toast:success("Operation completed")
window.toast:warning("Low disk space")
window.toast:error("Network failed")

-- 完整自定义
local id = window.toast:show{
    message    = "File uploaded successfully",
    title      = "Upload",
    level      = "success",        -- info / success / warning / error
    durationMs = 5000,             -- 0 = 持久(不自动关闭)
    position   = "top-right",      -- 6 选项,见 §4
    closable   = true,             -- 显示 × 关闭按钮
    icon       = "✓"               -- 可选,自定义 icon
}

-- 提前关闭
window.toast:dismiss(id)
window.toast:dismissAll()

-- 查询
print(window.toast:count())  -- 当前活跃数量

-- 配置(全局)
window.toast:setLeaveDurationMs(500)    -- 离场动画时长
window.toast:setInjectDefaultStyles(false)  -- 不注入 UA 默认 CSS

§3 C++ API

#include <kui/dom/toast_manager.h>

auto& tm = doc->toastManager();
tm.info("Saved");
tm.success("Operation done");
tm.warning("Low memory", 4000);
tm.error("Critical error", 5000);

ToastConfig cfg;
cfg.message    = "File uploaded";
cfg.title      = "Upload";
cfg.level      = ToastLevel::Success;
cfg.durationMs = 5000;
cfg.position   = "top-right";
u32 id = tm.show(cfg);

tm.dismiss(id);
tm.dismissAll();
